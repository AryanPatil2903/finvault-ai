"""

def gen_uuid() -> str:
    """Generate a new UUID4 string â€” used as default primary key."""
    return str(uuid.uuid4())


# â”€â”€ User â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
class User(Base):
    """Application user â€” owns all financial records."""

    __tablename__ = "users"

    id: str = Column(UUID(as_uuid=False), primary_key=True, default=gen_uuid)
    email: str = Column(String(255), unique=True, nullable=False, index=True)
    password_hash: str = Column(String(255), nullable=False)
    full_name: str = Column(String(255), nullable=False)
    phone: Optional[str] = Column(String(20), nullable=True)
    currency: str = Column(String(10), default="INR")
    monthly_income: float = Column(Float, default=0.0)  # For FOIR calculation
    is_active: bool = Column(Boolean, default=True)
    created_at: datetime = Column(DateTime(timezone=True), server_default=func.now())
    updated_at: datetime = Column(
        DateTime(timezone=True), server_default=func.now(), onupdate=func.now()
    )

    # â”€â”€ Relationships â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
    expenses: List["Expense"] = relationship("Expense", back_populates="user", cascade="all, delete-orphan")
    incomes: List["Income"] = relationship("Income", back_populates="user", cascade="all, delete-orphan")
    loans: List["Loan"] = relationship("Loan", back_populates="user", cascade="all, delete-orphan")
    budgets: List["Budget"] = relationship("Budget", back_populates="user", cascade="all, delete-orphan")
    ai_conversations: List["AIConversation"] = relationship("AIConversation", back_populates="user", cascade="all, delete-orphan")

class Expense(Base):

    __tablename__ = "expenses"

"""
Async SQLAlchemy engine, session factory, and base model setup.
All database I/O in this application uses async/await throughout.
"""

from typing import AsyncGenerator

from sqlalchemy.ext.asyncio import (
    AsyncEngine,
    AsyncSession,
    async_sessionmaker,
    create_async_engine,
)
from sqlalchemy.orm import DeclarativeBase

from config.settings import settings

# â”€â”€ Declarative base shared by all ORM models â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
class Base(DeclarativeBase):
    """Parent class for all SQLAlchemy models."""
    pass


# â”€â”€ Engine â€” created once at module import â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
engine: AsyncEngine = create_async_engine(
    settings.DATABASE_URL,
    pool_size=settings.DB_POOL_SIZE,
    max_overflow=settings.DB_MAX_OVERFLOW,
    echo=settings.DB_ECHO,
    pool_pre_ping=True,          # Verify connection health before use
    pool_recycle=3600,           # Recycle connections older than 1 hour
)

# â”€â”€ Session factory â€” use as async context manager â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
AsyncSessionLocal: async_sessionmaker[AsyncSession] = async_sessionmaker(
    bind=engine,
    class_=AsyncSession,
    expire_on_commit=False,      # Prevent lazy-load errors after commit
    autocommit=False,
    autoflush=False,
)


async def get_db() -> AsyncGenerator[AsyncSession, None]:
    """
    FastAPI dependency that yields a database session per request.
    Commits on success, rolls back on any exception.
    """
    async with AsyncSessionLocal() as session:
        try:
            yield session
            await session.commit()
        except Exception:
            await session.rollback()
            raise
        finally:
            await session.close()


async def init_db() -> None:
    """
    Create all tables defined in models on application startup.
    In production, use Alembic migrations instead.
    """
    # Import models so SQLAlchemy registers them before create_all
    from database import models  # noqa: F401

    async with engine.begin() as conn:
        await conn.run_sync(Base.metadata.create_all)


async def dispose_db() -> None:
    """Gracefully close all connections in the pool on shutdown."""
    await engine.dispose()
_is_sqlite = settings.DATABASE_URL.startswith("sqlite")
_engine_kwargs = {"echo": settings.DB_ECHO}
if not _is_sqlite:
    _engine_kwargs.update(
        pool_size=settings.DB_POOL_SIZE,
        max_overflow=settings.DB_MAX_OVERFLOW,
        pool_pre_ping=True,
        pool_recycle=3600,
    )

engine: AsyncEngine = create_async_engine(settings.DATABASE_URL, **_engine_kwargs)

        payment_frequency=body.payment_frequency,
        disbursement_date=body.disbursement_date,
        first_emi_date=body.first_emi_date,
        emi_due_day=body.emi_due_day,

@router.post("/{loan_id}/payments", response_model=LoanPaymentResponse, status_code=201)
async def record_emi_payment(
    loan_id: str,
    body: LoanPaymentCreateRequest,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
) -> LoanPayment:
    """
    Record an EMI payment against a loan.
    Automatically updates loan outstanding balance and EMI counters.
    """
    result = await db.execute(
        select(Loan).where(Loan.id == loan_id, Loan.user_id == current_user.id)
    )
    loan = result.scalar_one_or_none()
    if not loan:
        raise HTTPException(status_code=404, detail="Loan not found.")

    payment = LoanPayment(
        loan_id=loan_id,
        emi_number=body.emi_number,
        payment_date=body.payment_date,
        due_date=body.due_date,
        amount_paid=body.amount_paid,
        principal_component=body.principal_component,
        interest_component=body.interest_component,
        outstanding_after_payment=body.outstanding_after_payment,
        payment_method=body.payment_method,
        transaction_id=body.transaction_id,
        receipt_number=body.receipt_number,
        is_prepayment=body.is_prepayment,
        prepayment_amount=body.prepayment_amount,
        late_fee_paid=body.late_fee_paid,
        notes=body.notes,
    )
    db.add(payment)

    # Update loan counters
    loan.outstanding_amount = body.outstanding_after_payment
    loan.total_emis_paid += 1
    loan.total_emis_remaining = loan.tenure_months - loan.total_emis_paid
    if loan.outstanding_amount <= 0:
        loan.status = "Closed"

    await db.flush()
    return payment

@router.post("/{loan_id}/prepayment-impact")
async def prepayment_impact(
    loan_id: str,
    prepayment_amount: float,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
) -> Dict[str, Any]:
    """
    Calculate the impact of making a lump-sum prepayment on this loan.
    Returns: EMIs saved, interest saved, new tenure.
    """
    result = await db.execute(
        select(Loan).where(Loan.id == loan_id, Loan.user_id == current_user.id)
    )
    loan = result.scalar_one_or_none()
    if not loan:
        raise HTTPException(status_code=404, detail="Loan not found.")

    if prepayment_amount <= 0:
        raise HTTPException(status_code=400, detail="Prepayment amount must be positive.")

    scenario = calculate_prepayment_impact(
        current_outstanding=loan.outstanding_amount,
        annual_interest_rate=loan.annual_interest_rate,
        emis_remaining=loan.total_emis_remaining or loan.tenure_months,
        prepayment_amount=prepayment_amount,
        first_remaining_emi_date=date.today(),
        method=loan.interest_method,
    )

    prepayment_penalty = round(
        prepayment_amount * (loan.prepayment_penalty_percent / 100), 2
    )

    return {
        "loan_id": loan_id,
        "prepayment_amount": prepayment_amount,
        "prepayment_penalty": prepayment_penalty,
        "net_prepayment": prepayment_amount + prepayment_penalty,
        "original_emis_remaining": scenario.original_emis_remaining,
        "new_emis_remaining": scenario.new_emis_remaining,
        "emis_saved": scenario.emis_saved,
        "interest_saved": scenario.interest_saved,
        "new_outstanding": scenario.new_outstanding,
        "is_worth_it": scenario.interest_saved > prepayment_penalty,
    }

    )
    row = schedule[loan.total_emis_paid]  # Next unpaid EMI

    payment = LoanPayment(
        loan_id=loan_id,
        emi_number=row.emi_number,
        payment_date=date.today(),
        due_date=row.due_date,
        amount_paid=row.emi_amount,
        principal_component=row.principal_component,
        interest_component=row.interest_component,
        outstanding_after_payment=row.outstanding_after,
        payment_method=payment_method,
    )
    db.add(payment)

    loan.outstanding_amount = row.outstanding_after
    loan.total_emis_paid += 1
    loan.total_emis_remaining = loan.tenure_months - loan.total_emis_paid
    if loan.outstanding_amount <= 0:
        loan.status = "Closed"

    await db.flush()
    return payment
